> Módulo 14: Manejo de Errores

No dejar que tu programa se caiga.

try/except


try:
    numero = int(input("Número: "))
    resultado = 10 / numero
    print(f"Resultado: {resultado}")
except ValueError:
    print("No es un número válido")
except ZeroDivisionError:
    print("No se puede dividir entre cero")

Capturar el error


try:
    archivo = open("no_existe.txt")
except FileNotFoundError as e:
    print(f"Error: {e}")
    print("El archivo no existe")

finally — Siempre se ejecuta


try:
    archivo = open("datos.txt")
    contenido = archivo.read()
except FileNotFoundError:
    print("No se encontró el archivo")
finally:
    print("Esto se ejecuta siempre, haya error o no")

Crear excepciones propias


def dividir(a, b):
    if b == 0:
        raise ValueError("No se puede dividir entre cero")
    return a / b

try:
    print(dividir(10, 0))
except ValueError as e:
    print(f"Error: {e}")

Excepciones personalizadas


class EdadInvalidaError(Exception):
    def __init__(self, edad, mensaje="Edad no válida"):
        self.edad = edad
        self.mensaje = mensaje
        super().__init__(self.mensaje)

def registrar_edad(edad):
    if edad < 0 or edad > 150:
        raise EdadInvalidaError(edad)
    print(f"Edad registrada: {edad}")

try:
    registrar_edad(-5)
except EdadInvalidaError as e:
    print(f"Error: {e.edad} — {e.mensaje}")

Manejo de errores con archivos


import json

def cargar_seguro(archivo):
    try:
        with open(archivo, "r") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"Archivo {archivo} no encontrado — creando nuevo")
        return {}
    except json.JSONDecodeError:
        print(f"Archivo {archivo} corrupto — creando nuevo")
        return {}
    except PermissionError:
        print(f"Sin permisos para leer {archivo}")
        return {}

Ejemplos prácticos

Calculadora segura


def calculadora():
    while True:
        try:
            num1 = float(input("Número 1 (0 para salir): "))
            if num1 == 0:
                break
            
            op = input("Operación (+, -, *, /): ")
            num2 = float(input("Número 2: "))
            
            if op == "+":
                print(f"Resultado: {num1 + num2}")
            elif op == "-":
                print(f"Resultado: {num1 - num2}")
            elif op == "*":
                print(f"Resultado: {num1 * num2}")
            elif op == "/":
                if num2 == 0:
                    print("Error: No se puede dividir entre cero")
                else:
                    print(f"Resultado: {num1 / num2}")
            else:
                print("Operación no válida")
        except ValueError:
            print("Por favor ingresa números válidos")

calculadora()

Registro de usuario con validación


def registrar_usuario():
    while True:
        try:
            nombre = input("Nombre: ").strip()
            if len(nombre) < 2:
                raise ValueError("El nombre debe tener al menos 2 caracteres")
            
            edad = int(input("Edad: "))
            if edad < 0 or edad > 150:
                raise ValueError("Edad debe ser entre 0 y 150")
            
            email = input("Email: ").strip()
            if "@" not in email:
                raise ValueError("Email no válido")
            
            print(f"✅ Usuario registrado: {nombre}, {edad} años, {email}")
            break
            
        except ValueError as e:
            print(f"❌ Error: {e}. Intenta de nuevo.")

registrar_usuario()

import json
def cargar_json_seguro(ruta):
    try:
        with open(ruta, "r") as f:
            return json.load(f), None
    except FileNotFoundError:
        return None, "Archivo no encontrado"
    except json.JSONDecodeError:
        return None, "Formato JSON inválido"
