Herencia

Una clase hija hereda todo de la clase padre:


class Animal:
    def __init__(self, nombre, sonido):
        self.nombre = nombre
        self.sonido = sonido
    
    def hablar(self):
        return f"{self.nombre} hace {self.sonido}"

class Perro(Animal):
    def __init__(self, nombre):
        super().__init__(nombre, "guau")
    
    def fetch(self):
        return f"{self.nombre} trae la pelota"

class Gato(Animal):
    def __init__(self, nombre):
        super().__init__(nombre, "miau")

perro = Perro("Rex")
gato = Gato("Michi")

print(perro.hablar())   # Rex hace guau
print(perro.fetch())    # Rex trae la pelota
print(gato.hablar())    # Michi hace miau

Polimorfismo

Métodos con el mismo nombre pero comportamiento diferente:


class Forma:
    def area(self):
        raise NotImplementedError("Subclase debe implementar area()")
    
    def describir(self):
        return f"{self.__class__.__name__} con área {self.area():.2f}"

class Circulo(Forma):
    def __init__(self, radio):
        self.radio = radio
    
    def area(self):
        return 3.14159 * self.radio ** 2

class Rectangulo(Forma):
    def __init__(self, base, altura):
        self.base = base
        self.altura = altura
    
    def area(self):
        return self.base * self.altura

# Polimorfismo — misma interfaz, diferente comportamiento
formas = [Circulo(5), Rectangulo(4, 6), Circulo(3)]
for forma in/formas):
    print(forma.describir())

Métodos especiales (Dunder/Magic methods)


class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, otro):
        return Vector(self.x + otro.x, self.y + otro.y)
    
    def __mul__(self, escalar):
        return Vector(self.x * escalar, self.y * escalar)
    
    def __eq__(self, otro):
        return self.x == otro.x and self.y == otro.y
    
    def __len__(self):
        return int((self.x**2 + self.y**2)**0.5)
    
    def __str__(self):
        return f"({self.x}, {self.y})"

v1 = Vector(2, 3)
v2 = Vector(4, 1)

print(v1 + v2)      # (6, 4)
print(v1 * 3)       # (6, 9)
print(v1 == v2)     # False
print(len(v1))      # 3 (magnitud)

Propiedades (@property)


class Temperatura:
    def __init__(self, celsius):
        self._celsius = celsius
    
    @property
    def celsius(self):
        return self._celsius
    
    @celsius.setter
    def celsius(self, valor):
        if valor < -273.15:
            raise ValueError("Temperatura menor al cero absoluto")
        self._celsius = valor
    
    @property
    def fahrenheit(self):
        return self._celsius * 9/5 + 32

t = Temperatura(100)
print(t.fahrenheit)    # 212.0
t.celsius = 0          # Setter
print(t.fahrenheit)    # 32.0

Clases abstractas


from abc import ABC, abstractmethod

class Vehiculo(ABC):
    @abstractmethod
    def arrancar(self):
        pass
    
    @abstractmethod
    def frenar(self):
        pass
    
    def describir(self):
        return f"Soy un {self.__class__.__name__}"

class Moto(Vehiculo):
    def arrancar(self):
        return "Moto arrancando"
    
    def frenar(self):
        return "Moto frenando"

# v = Vehiculo()  # ❌ Error — es abstracta
m = Moto()
print(m.arrancar())

Ejemplos prácticos

Sistema de empleados


class Empleado:
    def __init__(self, nombre, salario):
        self.nombre = nombre
        self.salario = salario
    
    def calcular_bono(self):
        return self.salario * 0.10
    
    def __str__(self):
        return f"{self.nombre} — Bono: ${self.calcular_bono():.2f}"

class Gerente(Empleado):
    def __init__(self, nombre, salario, departamento):
        super().__init__(nombre, salario)
        self.departamento = departamento
    
    def calcular_bono(self):
        return self.salario * 0.20  # Gerentes ganan más bono

class Desarrollador(Empleado):
    def __init__(self, nombre, salario, lenguajes):
        super().__init__(nombre, salario)
        self.lenguajes = lenguajes
    
    def calcular_bono(self):
        return self.salario * 0.15 + len(self.lenguajes) * 500

empleados = [
    Gerente("Ana", 5000, "TI"),
    Desarrollador("Carlos", 3000, ["Python", "JS"]),
    Empleado("Luis", 2500),
]

for emp in empleados:
    print(emp)  # Polimorfismo — cada uno calcula su bono diferente

# Ejercicio 2
class Cuenta:
    def __init__(self, titular, saldo=0):
        self.titular = titular
        self.saldo = saldo

class CuentaAhorro(Cuenta):
    def __init__(self, titular, saldo=0, tasa=0.02):
        super().__init__(titular, saldo)
        self.tasa = tasa
    
    def aplicar_interes(self):
        self.saldo += self.saldo * self.tasa

class CuentaCorriente(Cuenta):
    def __init__(self, titular, saldo=0, limite=500):
        super().__init__(titular, saldo)
        self.limite = limite
    
    def retirar(self, monto):
        if monto <= self.saldo + self.limite:
            self.saldo -= monto
            return True
        return False
