> Módulo 09: Diccionarios

Pares de clave-valor — como un diccionario real.

Crear diccionarios


persona = {
    "nombre": "Aaron",
    "edad": 25,
    "ciudad": "CDMX"
}

vacio = {}

Acceder a valores


persona = {"nombre": "Aaron", "edad": 25}

print(persona["nombre"])     # Aaron
print(persona.get("edad"))   # 25
print(persona.get("email", "no disponible"))  # Valor por defecto si no existe

Modificar y agregar


persona = {"nombre": "Aaron", "edad": 25}

persona["edad"] = 26           # Modificar
persona["email"] = "a@b.com"   # Agregar nuevo
del persona["email"]           # Eliminar

Métodos útiles


persona = {"nombre": "Aaron", "edad": 25, "ciudad": "CDMX"}

print(persona.keys())     # dict_keys(['nombre', 'edad', 'ciudad'])
print(persona.values())   # dict_values(['Aaron', 25, 'CDMX'])
print(persona.items())    # dict_items([('nombre', 'Aaron'), ...])

print("nombre" in persona)  # True (verificar si existe la clave)

Iterar diccionarios


persona = {"nombre": "Aaron", "edad": 25, "ciudad": "CDMX"}

# Por clave
for clave in persona:
    print(clave)

# Por valor
for clave, valor in persona.items():
    print(f"{clave}: {valor}")

Diccionarios anidados


escuela = {
    "clase_A": {
        "maestro": "Prof. García",
        "alumnos": 30
    },
    "clase_B": {
        "maestro": "Prof. López",
        "alumnos": 28
    }
}

print(escuela["clase_A"]["maestro"])  # Prof. García

List comprehension con diccionarios


# Cuadrados del 1 al 5
cuadrados = {x: x**2 for x in range(1, 6)}
print(cuadrados)  # {1: 1, 2: 4, 3: 9, 4: 16, 5: 25}

# Filtrar
mayores = {k: v for k, v in cuadrados.items() if v > 10}
print(mayores)  # {4: 16, 5: 25}

Ejemplos prácticos

Contador de palabras


texto = "el gato come pescado el gato duerme"
palabras = texto.split()

contador = {}
for palabra in palabras:
    contador[palabra] = contador.get(palabra, 0) + 1

print(contador)  # {'el': 2, 'gato': 2, 'come': 1, 'pescado': 1, 'duerme': 1}

Agenda de contactos


contactos = {}

while True:
    accion = input("(a)gregar, (b)uscar, (l)istar, (s)alir: ")
    
    if accion == "a":
        nombre = input("Nombre: ")
        telefono = input("Teléfono: ")
        contactos[nombre] = telefono
        print("Guardado")
    elif accion == "b":
        nombre = input("Nombre: ")
        print(contactos.get(nombre, "No encontrado"))
    elif accion == "l":
        for nombre, tel in contactos.items():
            print(f"{nombre}: {tel}")
    elif accion == "s":
        break

Ejercicios

  • Crea un diccionario con 3 productos (nombre → precio). Imprime el más caro
  • Invierte un diccionario (clave → valor se convierte en valor → clave)
  • Cuenta las letras de una palabra y guárdalas en un diccionario
  • Fusiona dos diccionarios (si hay clave repetida, suma los valores)

    • Soluciones

    
# Ejercicio 1
productos = {"laptop": 800, "mouse": 25, "teclado": 45}
mas_caro = max(productos, key=productos.get)
print(f"El más caro es {mas_caro} con ${productos[mas_caro]}")

# Ejercicio 2
original = {"a": 1, "b": 2, "c": 3}
invertido = {v: k for k, v in original.items()}
print(invertido)  # {1: 'a', 2: 'b', 3: 'c'}

# Ejercicio 3
palabra = "murcielago"
contador = {}
for letra in palabra:
    contador[letra] = contador.get(letra, 0) + 1
print(contador)

# Ejercicio 4
dict1 = {"a": 1, "b": 2}
dict2 = {"b": 3, "c": 4}
fusionado = dict1.copy()
for k, v in dict2.items():
    fusionado[k] = fusionado.get(k, 0) + v
print(fusionado)  # {'a': 1, 'b': 5, 'c': 4}
