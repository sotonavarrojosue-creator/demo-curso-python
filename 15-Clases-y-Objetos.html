> Módulo 15: Clases y Objetos

Programación Orientada a Objetos — modelar el mundo real.

¿Qué es una clase?

Una clase es un plano/plantilla. Un objeto es una instancia de esa clase.


class Persona:
    def __init__(self, nombre, edad):
        self.nombre = nombre
        self.edad = edad
    
    def saludar(self):
        return f"Hola, soy {nombre} y tengo {self.edad} años"

# Crear objetos
aaron = Persona("Aaron", 25)
beto = Persona("Beto", 30)

print(aaron.saludar())
print(beto.saludar())

`__init__` — Constructor

Se ejecuta automáticamente al crear un objeto:


class Auto:
    def __init__(self, marca, color, año):
        self.marca = marca
        self.color = color
        self.año = año
        self.encendido = False  # Valor por defecto
    
    def encender(self):
        self.encendido = True
        return f"{self.marca} encendido"

mi_auto = Auto("Toyota", "Rojo", 2020)
print(mi_auto.encender())  # Toyota encendido

Métodos

Funciones dentro de una clase que operan sobre los datos del objeto:


class Circulo:
    def __init__(self, radio):
        self.radio = radio
    
    def area(self):
        return 3.14159 * self.radio ** 2
    
    def perimetro(self):
        return 2 * 3.14159 * self.radio
    
    def describir(self):
        return f"Círculo de radio {self.radio}, área {self.area():.2f}"

c = Circulo(5)
print(c.area())        # 78.54
print(c.describir())   # Círculo de radio 5, área 78.54

Atributos de clase vs instancia


class Producto:
    # Atributo de clase — compartido por todos los objetos
    impuesto = 0.16
    
    def __init__(self, nombre, precio):
        # Atributos de instancia — únicos por objeto
        self.nombre = nombre
        self.precio = precio
    
    def precio_con_impuesto(self):
        return self.precio * (1 + self.impuesto)

p1 = Producto("Laptop", 800)
p2 = Producto("Mouse", 25)

print(p1.precio_con_impuesto())  # 928.0
print(p2.precio_con_impuesto())  # 29.0

`__str__` y `__repr__`


class Persona:
    def __init__(self, nombre, edad):
        self.nombre = nombre
        self.edad = edad
    
    def __str__(self):
        return f"{self.nombre}, {self.edad} años"
    
    def __repr__(self):
        return f"Persona('{self.nombre}', {self.edad})"

p = Persona("Aaron", 25)
print(p)         # Aaron, 25 años (usa __str__)
print(repr(p))   # Persona('Aaron', 25) (usa __repr__)

Encapsulamiento


class CuentaBancaria:
    def __init__(self, titular, saldo=0):
        self.titular = titular
        self.__saldo = saldo  # Privado (doble guión bajo)
    
    def depositar(self, monto):
        if monto > 0:
            self.__saldo += monto
            return True
        return False
    
    def retirar(self, monto):
        if 0 < monto <= self.__saldo:
            self.__saldo -= monto
            return True
        return False
    
    def ver_saldo(self):
        return self.__saldo

cuenta = CuentaBancaria("Aaron", 1000)
cuenta.depositar(500)
cuenta.retirar(200)
print(cuenta.ver_saldo())  # 1300
# print(cuenta.__saldo)    # ❌ Error — es privado

Ejemplos prácticos

Clase Rectángulo


class Rectangulo:
    def __init__(self, base, altura):
        self.base = base
        self.altura = altura
    
    def area(self):
        return self.base * self.altura
    
    def perimetro(self):
        return 2 * (self.base + self.altura)
    
    def es_cuadrado(self):
        return self.base == self.altura
    
    def __str__(self):
        return f"Rectángulo({self.base}x{self.altura})"

r = Rectangulo(10, 5)
print(f"Área: {r.area()}")        # 50
print(f"¿Cuadrado? {r.es_cuadrado()}")  # False

Clase Estudiante


class Estudiante:
    def __init__(self, nombre):
        self.nombre = nombre
        self.notas = []
    
    def agregar_nota(self, nota):
        self.notas.append(nota)
    
    def promedio(self):
        if not self.notas:
            return 0
        return sum(self.notas) / len(self.notas)
    
    def esta_aprobado(self):
        return self.promedio() >= 70
    
    def __str__(self):
        estado = "Aprobado" if self.esta_aprobado() else "Reprobado"
        return f"{self.nombre}: {self.promedio():.1f} ({estado})"

est = Estudiante("Aaron")
est.agregar_nota(85)
est.agregar_nota(92)
est.agregar_nota(78)
print(est)  # Aaron: 85.0 (Aprobado)

# Ejercicio 1
class Coche:
    def __init__(self, marca, modelo, año):
        self.marca = marca
        self.modelo = modelo
        self.año = año
        self.encendido = False
        self.velocidad = 0
    
    def arrancar(self):
        self.encendido = True
        return f"{self.marca} arrancó"
    
    def frenar(self):
        self.velocidad = 0
        return "Frenando..."
    
    def acelerar(self, incremento):
        if self.encendido:
            self.velocidad += incremento
            return f"Velocidad: {self.velocidad} km/h"
        return "Primero arrancá"

# Ejercicio 2
class Cuenta:
    def __init__(self, titular, saldo=0):
        self.titular = titular
        self.__saldo = saldo
    
    def depositar(self, monto):
        if monto > 0:
            self.__saldo += monto
            return f"Depósito: ${monto}. Saldo: ${self.__saldo}"
        return "Monto inválido"
    
    def retirar(self, monto):
        if 0 < monto <= self.__saldo:
            self.__saldo -= monto
            return f"Retiro: ${monto}. Saldo: ${self.__saldo}"
        return "Fondos insuficientes"
    
    def ver_saldo(self):
        return f"Saldo: ${self.__saldo}"

# Ejercicio 4
class Calculadora:
    def __init__(self):
        self.historial = []
    
    def operar(self, a, b, op):
        if op == "+": r = a + b
        elif op == "-": r = a - b
        elif op == "*": r = a * b
        elif op == "/": r = a / b if b != 0 else "Error"
        self.historial.append(f"{a} {op} {b} = {r}")
        return r
    
    def ver_historial(self):
        for item in self.historial:
            print(item)
